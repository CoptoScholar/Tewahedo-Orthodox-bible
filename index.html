async function openChapter(chapter){ selectedChapter=chapter; document.getElementById("readingBookTitle").textContent= selectedBook.amharic+" — "+selectedBook.english; document.getElementById("readingChapterTitle").textContent= "ምዕራፍ "+chapter; const content=document.getElementById("readingContent"); content.innerHTML=""; if(selectedBook.old){ const p=document.createElement("div"); p.className="reader-placeholder"; p.textContent="የምዕራፉን ጽሑፍ በመጫን ላይ..."; content.appendChild(p); const sourceId=oldSourceBooks[selectedBook.english]; if(!sourceId){ p.textContent="የዚህ መጽሐፍ ጽሑፍ ገና አልተገናኘም።"; setScreen("reading"); return; } try{ const url= "https://www.wordproject.org/bibles/am/"+ sourceId+"/"+chapter+".htm"; const response=await fetch(url); if(!response.ok) throw new Error("Unable to load chapter"); const html=await response.text(); const parsed=new DOMParser().parseFromString(html,"text/html"); const mainText=parsed.querySelector( "body" ); if(!mainText) throw new Error("Chapter text not found"); const text=mainText.innerText.trim(); content.innerHTML=""; const verse=document.createElement("div"); verse.className="verse"; const verseText=document.createElement("span"); verseText.className="verse-text"; verseText.textContent=text; verse.appendChild(verseText); content.appendChild(verse); }catch(error){ content.innerHTML=""; const message=document.createElement("div"); message.className="reader-placeholder"; message.textContent= "ጽሑፉን መጫን አልተቻለም። እባክዎ እንደገና ይሞክሩ።"; content.appendChild(message); const link=document.createElement("a"); link.className="source-link"; link.href= "https://www.wordproject.org/bibles/am/"+ sourceId+"/"+chapter+".htm"; link.target="_blank"; link.rel="noopener noreferrer"; link.textContent="ምዕራፉን በምንጩ ይክፈቱ"; content.appendChild(link); } }else{ const verses=scriptureData[selectedBook.id]?.[chapter]; if(verses&&verses.length){ verses.sort((a,b)=>a.number-b.number).forEach(verse=>{ const row=document.createElement("div"); row.className="verse"; const number=document.createElement("span"); number.className="verse-number"; number.textContent=verse.number; const text=document.createElement("span"); text.className="verse-text"; text.textContent=verse.text; row.append(number,text); content.appendChild(row); }); }else{ const p=document.createElement("div"); p.className="reader-placeholder"; p.textContent="የዚህ ምዕራፍ ጽሑፍ አልተገኘም።"; content.appendChild(p); } } setScreen("reading"); }
